import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import accuracy_score

# Loading the diabetes dataset
parkinsons_data = pd.read_csv(r"parkinsons.csv")

# Printing the first 5 rows of the dataset
print(parkinsons_data.head())

print(parkinsons_data.shape)

print(parkinsons_data.info())

print(parkinsons_data.isnull().sum())

# Statistical measures about the data
print(parkinsons_data.describe())

# Checking the distribution of Target Variable
print(parkinsons_data['status'].value_counts())

# Calculate the mean for numeric columns only
numeric_data = parkinsons_data.select_dtypes(include=np.number)
print(numeric_data.groupby('status').mean())

# Separating the data and labels
X = parkinsons_data.drop(columns=['name', 'status'], axis=1)
Y = parkinsons_data['status']

print(X)

print(Y)

# Splitting the dataset into training and testing sets
X_train, X_test, Y_train, Y_test = train_test_split(X, Y, test_size=0.2, random_state=2)
print(X.shape, X_train.shape, X_test.shape)

# Using KNN algorithm
model = KNeighborsClassifier(n_neighbors=5)

# Training the KNN model with training data
model.fit(X_train, Y_train)

# Accuracy score on the training data
X_train_prediction = model.predict(X_train)
training_data_accuracy = accuracy_score(Y_train, X_train_prediction)
print('Accuracy score of training data : ', training_data_accuracy)

# Accuracy score on the test data
X_test_prediction = model.predict(X_test)
test_data_accuracy = accuracy_score(Y_test, X_test_prediction)
print('Accuracy score of test data : ', test_data_accuracy)

# Input data for prediction
input_data = (197.07600, 206.89600, 192.05500, 0.00289, 0.00001, 0.00166, 0.00168, 0.00498, 0.01098, 0.09700, 0.00563, 0.00680, 0.00802, 0.01689, 0.00339, 26.77500, 0.422229, 0.741367, -7.348300, 0.177551, 1.743867, 0.085569)

# Changing input data to a numpy array
input_data_as_numpy_array = np.asarray(input_data)

# Reshape the numpy array
input_data_reshaped = input_data_as_numpy_array.reshape(1, -1)

# Making a prediction
prediction = model.predict(input_data_reshaped)
print(prediction)

if (prediction[0] == 0):
    print("The Person does not have Parkinson's Disease")
else:
    print("The Person has Parkinson's Disease")
